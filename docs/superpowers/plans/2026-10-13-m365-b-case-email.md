# Microsoft 365 — Part B: Case Email Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Email from Outlook folders that are linked to a case, plus threads linked one at a time, shows up in a new "อีเมล" tab on the case. Users can read the full sanitized HTML, download attachments, save attachments into case documents, and see the emails on the case timeline.

**Architecture:** A new `CaseExternalLink` table (kinds `OUTLOOK_FOLDER` now, `DRIVE_FOLDER` for Phase D) ties a mail folder in the linking user's own mailbox to a case. A pure matcher, `suggestCaseMatches` (in `case-matching.ts`), suggests a case from the folder name. Phase D reuses it for drive folders. Nothing is linked until a person confirms. `CaseFolderSyncService` runs a Graph delta query per linked folder: on a 5-minute cron, on "ซิงก์ตอนนี้", and once (fire-and-forget) right after linking, which backfills the whole folder. Each message is written through Phase 0's `OutlookSyncService.upsertGraphMessage` with `caseId` set. Single threads are linked and unlinked through `CaseEmailService` in the email-intake module. That service also serves the case tab, enforcing `getEmailThreadFilterForUser`, and saves attachments through `DocumentsService.createFromBuffer`.

**Tech Stack:** NestJS + Prisma (Postgres) in `apps/api`, `@nestjs/schedule` cron, Microsoft Graph via Phase 0's `GraphHttp` / `OutlookGraphClient`, Next.js (app router, client components) in `apps/web`, Jest for API tests, and `node --test` for web lib tests.

**Spec:** `docs/superpowers/specs/2026-10-13-microsoft365-integration-design.md`. Scope is **ส่วน B — อีเมลเข้าสำนวน** only.

**Depends on:** `docs/superpowers/plans/2026-10-13-m365-0-foundation-calendar-teams.md`. Run Phase 0 first.

```ts
import type { Prisma } from '../generated/prisma';

/**
 * Exact-rule matching of an external name (Outlook folder, OneDrive folder,
 * email subject) to cases — no AI. Callers only ever SHOW these suggestions;
 * a person confirms every link (spec ส่วน B / D).
 */
export interface MatchableCase {
  id: string;
  ownRef: string;
  title: string;
  clientName: string | null;
  caseNumbers: string[];
}

export type CaseMatchReason = 'OWN_REF' | 'CASE_NUMBER' | 'CLIENT_NAME';

export interface CaseMatchSuggestion {
  caseId: string;
  ownRef: string;
  title: string;
  reason: CaseMatchReason;
  matchedText: string;
}

const REASON_RANK: Record<CaseMatchReason, number> = { OWN_REF: 0, CASE_NUMBER: 1, CLIENT_NAME: 2 };
const MIN_TOKEN = 3;
const NOT_WORD_BEFORE = '(?<![\\p{L}\\p{N}])';
const NOT_WORD_AFTER = '(?![\\p{L}\\p{N}])';
/** "ผบ.1234/2567" → 1234 / 2567. Thai numerals are not handled (case numbers are stored with ASCII digits). */
const CASE_NUMBER_CORE = /(\d+)\s*\/\s*(\d{2,4})/;
/** Longest alternatives first (นางสาว before นาง, (มหาชน) before มหาชน). */
const NAME_AFFIXES =
  /บริษัท|บจก\.?|บมจ\.?|ห้างหุ้นส่วนจำกัด|ห้างหุ้นส่วนสามัญ|หจก\.?|\(มหาชน\)|มหาชน|จำกัด|นางสาว|นาง|นาย|co\.?,?\s*ltd\.?|company|limited|ltd\.?|inc\.?/giu;

export const MATCHABLE_CASE_SELECT = {
  id: true,
  ownRef: true,
  title: true,
  clientName: true,
  blackCaseNumber: true,
  redCaseNumber: true,
  client: { select: { name: true } },
  proceedings: { select: { blackCaseNumber: true, redCaseNumber: true } },
} satisfies Prisma.CaseSelect;

export type MatchableCaseRow = Prisma.CaseGetPayload<{ select: typeof MATCHABLE_CASE_SELECT }>;

export function toMatchableCase(row: MatchableCaseRow): MatchableCase {
  const numbers = [
    row.blackCaseNumber,
    row.redCaseNumber,
    ...row.proceedings.flatMap((p) => [p.blackCaseNumber, p.redCaseNumber]),
  ]
    .map((n) => n?.trim() ?? '')
    .filter((n) => n.length > 0);
  return {
    id: row.id,
    ownRef: row.ownRef,
    title: row.title,
    clientName: row.client?.name ?? row.clientName,
    caseNumbers: [...new Set(numbers)],
  };
}

function escapeRegex(value: string): string {
  return value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

function hasOwnRef(name: string, ownRef: string): boolean {
  const ref = ownRef.trim();
  if ([...ref].length < MIN_TOKEN) return false;
  return new RegExp(`${NOT_WORD_BEFORE}${escapeRegex(ref)}${NOT_WORD_AFTER}`, 'iu').test(name);
}

function hasCaseNumber(name: string, caseNumber: string): boolean {
  const core = CASE_NUMBER_CORE.exec(caseNumber);
  if (!core) return false;
  return new RegExp(`(?<!\\d)${core[1]}(?:\\s*[-/_.]\\s*|\\s+)${core[2]}(?!\\d)`).test(name);
}

function nameCore(value: string): string {
  return value.normalize('NFC').replace(NAME_AFFIXES, '').replace(/\s+/g, '').toLowerCase();
}

function hasClientName(name: string, clientName: string | null): boolean {
  if (!clientName) return false;
  const core = nameCore(clientName);
  // Thai names have no word breaks, so this is a plain substring test — the weakest signal, ranked last.
  return [...core].length >= MIN_TOKEN && nameCore(name).includes(core);
}

export function suggestCaseMatches(folderName: string, cases: MatchableCase[]): CaseMatchSuggestion[] {
  const name = folderName.normalize('NFC').trim();
  if (!name) return [];

  const suggestions: CaseMatchSuggestion[] = [];
  for (const candidate of cases) {
    const base = { caseId: candidate.id, ownRef: candidate.ownRef, title: candidate.title };
    if (hasOwnRef(name, candidate.ownRef)) {
      suggestions.push({ ...base, reason: 'OWN_REF', matchedText: candidate.ownRef });
      continue;
    }
    const caseNumber = candidate.caseNumbers.find((n) => hasCaseNumber(name, n));
    if (caseNumber) {
      suggestions.push({ ...base, reason: 'CASE_NUMBER', matchedText: caseNumber });
      continue;
    }
    if (candidate.clientName && hasClientName(name, candidate.clientName)) {
      suggestions.push({ ...base, reason: 'CLIENT_NAME', matchedText: candidate.clientName });
    }
  }
  return suggestions.sort(
    (a, b) => REASON_RANK[a.reason] - REASON_RANK[b.reason] || a.ownRef.localeCompare(b.ownRef),
  );
}
```

```ts
import { Injectable, Logger } from '@nestjs/common';
import { Cron, CronExpression } from '@nestjs/schedule';
import { PrismaService } from '../prisma/prisma.module';
import { CaseExternalLinkKind, MailboxConnectionStatus } from '../generated/prisma';
import { GraphApiError, GraphDeltaResult, GraphMessage, OutlookGraphClient } from './outlook-graph.client';
import { OutlookConnectionsService } from './outlook-connections.service';
import { OutlookSyncService } from './outlook-sync.service';

/** Delta reports deletions/moves-out as `{ id, '@removed': {...} }` — email already filed in the case stays there. */
const isRemoved = (message: GraphMessage) => '@removed' in message;

export function describeFolderSyncError(error: unknown): string {
  if (error instanceof GraphApiError && error.status === 404) {
    return 'ไม่พบ folder นี้ใน Outlook แล้ว (อาจถูกลบหรือย้าย) — ยกเลิกการผูกแล้วผูกใหม่จากหน้าคดี';
  }
  return `ซิงก์ไม่สำเร็จ: ${error instanceof Error ? error.message : String(error)}`;
}

/**
 * Pulls mail from Outlook folders linked to a case (CaseExternalLink kind
 * OUTLOOK_FOLDER) via a per-folder delta query and files every message into
 * that case through OutlookSyncService.upsertGraphMessage. The first run on a
 * fresh link (deltaLink null) is the full backfill.
 */
@Injectable()
export class CaseFolderSyncService {
  private readonly logger = new Logger(CaseFolderSyncService.name);
  // ponytail: in-process guard only — two API instances could still sync one link at once; messageKey dedup absorbs it bar a rare unique-constraint race. Move to a DB lease if the API is scaled out.
  private readonly inFlight = new Set<string>();
  private cronRunning = false;

  constructor(
    private prisma: PrismaService,
    private connections: OutlookConnectionsService,
    private graph: OutlookGraphClient,
    private mailSync: OutlookSyncService,
  ) {}

  async syncLink(linkId: string): Promise<{ messagesSynced: number }> {
    if (this.inFlight.has(linkId)) return { messagesSynced: 0 };
    this.inFlight.add(linkId);
    try {
      return await this.runLink(linkId);
    } finally {
      this.inFlight.delete(linkId);
    }
  }

  private async runLink(linkId: string): Promise<{ messagesSynced: number }> {
    const link = await this.prisma.caseExternalLink.findUniqueOrThrow({
      where: { id: linkId },
      include: { connection: { select: { mailboxAddress: true } } },
    });
    try {
      const accessToken = await this.connections.getValidAccessToken(link.connectionId);
      const ownMailbox = link.connection.mailboxAddress.toLowerCase();
      let cursor = link.deltaLink;
      let resetDone = false;
      let messagesSynced = 0;

      for (;;) {
        let page: GraphDeltaResult;
        try {
          page = await this.graph.getFolderDelta(accessToken, link.externalId, cursor);
        } catch (error) {
          if (error instanceof GraphApiError && error.status === 410 && cursor && !resetDone) {
            // Graph dropped the sync state (long downtime) — resync the whole folder; messageKey dedup keeps it idempotent.
            cursor = null;
            resetDone = true;
            continue;
          }
          throw error;
        }

        for (const message of page.messages) {
          if (isRemoved(message)) continue;
          const sender = message.from?.emailAddress?.address?.toLowerCase();
          const result = await this.mailSync.upsertGraphMessage({
            firmId: link.firmId,
            connectionId: link.connectionId,
            accessToken,
            message,
            direction: sender === ownMailbox ? 'OUTBOUND' : 'INBOUND',
            caseId: link.caseId,
            createThreadIfMissing: true,
          });
          if (result) messagesSynced += 1;
        }

        const next = page.nextDeltaLink ?? page.nextSkipLink;
        if (next) {
          cursor = next;
          // Saved per page so an interrupted first backfill resumes instead of starting over.
          await this.prisma.caseExternalLink.update({ where: { id: linkId }, data: { deltaLink: cursor } });
        }
        if (page.nextDeltaLink || !page.nextSkipLink) break;
      }

      await this.prisma.caseExternalLink.update({
        where: { id: linkId },
        data: { lastSyncedAt: new Date(), lastError: null },
      });
      return { messagesSynced };
    } catch (error) {
      await this.prisma.caseExternalLink
        .update({ where: { id: linkId }, data: { lastError: describeFolderSyncError(error).slice(0, 2000) } })
        .catch(() => undefined); // link may have been unlinked mid-sync
      throw error;
    }
  }

  async syncCase(caseId: string): Promise<{ messagesSynced: number; errors: string[] }> {
    const links = await this.prisma.caseExternalLink.findMany({
      where: { caseId, kind: CaseExternalLinkKind.OUTLOOK_FOLDER, connection: { status: MailboxConnectionStatus.ACTIVE } },
      select: { id: true, displayPath: true },
    });
    let messagesSynced = 0;
    const errors: string[] = [];
    for (const link of links) {
      try {
        messagesSynced += (await this.syncLink(link.id)).messagesSynced;
      } catch (error) {
        errors.push(`${link.displayPath}: ${describeFolderSyncError(error)}`);
      }
    }
    return { messagesSynced, errors };
  }

  @Cron(CronExpression.EVERY_5_MINUTES)
  async syncAllLinks(): Promise<void> {
    if (this.cronRunning) return;
    this.cronRunning = true;
    try {
      const links = await this.prisma.caseExternalLink.findMany({
        where: {
          kind: CaseExternalLinkKind.OUTLOOK_FOLDER,
          connection: { status: MailboxConnectionStatus.ACTIVE },
          case: { deletedAt: null },
        },
        select: { id: true },
      });
      for (const link of links) {
        try {
          await this.syncLink(link.id);
        } catch (error) {
          this.logger.warn(`Folder link ${link.id} sync failed: ${error}`);
        }
      }
    } finally {
      this.cronRunning = false;
    }
  }
}
```

```ts
import { Injectable } from '@nestjs/common';
import { GraphHttp } from './graph-http';

interface GraphMailFolder {
  id: string;
  displayName: string;
  childFolderCount: number;
}

export interface MailFolderEntry {
  id: string;
  displayName: string;
  displayPath: string;
  /** Inbox/Sent/Drafts/... — never linkable to a case (would pour the whole mailbox into it). */
  isSystem: boolean;
}

const FOLDER_QUERY = '$top=100&$select=id,displayName,childFolderCount';
const WELL_KNOWN = ['inbox', 'sentitems', 'drafts', 'deleteditems', 'junkemail', 'outbox', 'archive'] as const;
// ponytail: depth cap guards against pathological trees; raise if a firm nests case folders deeper than 6.
const MAX_DEPTH = 6;

@Injectable()
export class GraphMailFoldersClient {
  constructor(private http: GraphHttp) {}

  async listMailFolders(accessToken: string): Promise<MailFolderEntry[]> {
    const systemIds = await this.wellKnownFolderIds(accessToken);
    const out: MailFolderEntry[] = [];
    const walk = async (path: string, parentPath: string, depth: number): Promise<void> => {
      const folders = await this.http.paginate<GraphMailFolder>(accessToken, path);
      for (const folder of folders) {
        const displayPath = parentPath ? `${parentPath}/${folder.displayName}` : folder.displayName;
        out.push({ id: folder.id, displayName: folder.displayName, displayPath, isSystem: systemIds.has(folder.id) });
        if (folder.childFolderCount > 0 && depth < MAX_DEPTH) {
          await walk(`/me/mailFolders/${encodeURIComponent(folder.id)}/childFolders?${FOLDER_QUERY}`, displayPath, depth + 1);
        }
      }
    };
    await walk(`/me/mailFolders?${FOLDER_QUERY}`, '', 0);
    return out;
  }

  /** mailFolder has no wellKnownName in v1.0 — resolve the ids by their well-known aliases (missing ones, e.g. archive, just 404). */
  private async wellKnownFolderIds(accessToken: string): Promise<Set<string>> {
    const ids = await Promise.all(
```

```ts
import {
  BadRequestException,
  ConflictException,
  ForbiddenException,
  Injectable,
  Logger,
  NotFoundException,
} from '@nestjs/common';
import { AuthUser, hasFirmCapability } from '@lawfirm/shared';
import { PrismaService } from '../prisma/prisma.module';
import { CaseAccessService } from '../common/services/case-access.service';
import { CaseExternalLinkKind } from '../generated/prisma';
import { OutlookConnectionsService } from './outlook-connections.service';
import { GraphMailFoldersClient } from './graph/graph-mail-folders.client';
import { CaseFolderSyncService } from './case-folder-sync.service';
import { CaseMatchSuggestion, MATCHABLE_CASE_SELECT, suggestCaseMatches, toMatchableCase } from './case-matching';

export interface OutlookFolderRow {
  folderId: string;
  displayPath: string;
  link: { id: string; caseId: string; ownRef: string; title: string } | null;
  suggestions: CaseMatchSuggestion[];
}

const OUTLOOK_FOLDER = CaseExternalLinkKind.OUTLOOK_FOLDER;
const MAX_SUGGESTIONS_PER_FOLDER = 5;

@Injectable()
export class CaseFolderLinksService {
  private readonly logger = new Logger(CaseFolderLinksService.name);

  constructor(
    private prisma: PrismaService,
    private connections: OutlookConnectionsService,
    private folders: GraphMailFoldersClient,
    private caseAccess: CaseAccessService,
    private sync: CaseFolderSyncService,
  ) {}

  /** Links are always made from the caller's OWN mailbox — never someone else's. */
  private async ownConnection(user: AuthUser) {
    const connection = await this.connections.findActiveForUser(user.id);
    if (!connection) throw new BadRequestException('ยังไม่ได้เชื่อม Microsoft 365 — เชื่อมที่หน้าตั้งค่าก่อน');
    const accessToken = await this.connections.getValidAccessToken(connection.id);
    return { connection, accessToken };
  }

  async listFolders(user: AuthUser): Promise<OutlookFolderRow[]> {
    const { connection, accessToken } = await this.ownConnection(user);
    const [folders, caseRows, links] = await Promise.all([
      this.folders.listMailFolders(accessToken),
      this.prisma.case.findMany({ where: this.caseAccess.getCaseFilterForUser(user), select: MATCHABLE_CASE_SELECT }),
      this.prisma.caseExternalLink.findMany({
        where: { connectionId: connection.id, kind: OUTLOOK_FOLDER },
        select: { id: true, externalId: true, caseId: true, case: { select: { ownRef: true, title: true } } },
      }),
    ]);
    const cases = caseRows.map(toMatchableCase);
    const linkByFolder = new Map(links.map((link) => [link.externalId, link]));

    return folders
      .filter((folder) => !folder.isSystem)
      .map((folder) => {
        const link = linkByFolder.get(folder.id);
        return {
          folderId: folder.id,
          displayPath: folder.displayPath,
          link: link ? { id: link.id, caseId: link.caseId, ownRef: link.case.ownRef, title: link.case.title } : null,
          // ponytail: O(folders × cases) regex scan per request — fine for hundreds × low thousands; precompile/index if a firm outgrows it.
          suggestions: link ? [] : suggestCaseMatches(folder.displayName, cases).slice(0, MAX_SUGGESTIONS_PER_FOLDER),
        };
      });
  }

  /** The human "ยืนยัน" step. All-or-nothing: every item is validated before anything is written. */
  async linkFolders(user: AuthUser, items: { folderId: string; caseId: string }[]) {
    if (items.length === 0) return { linked: [] };
    const folderIds = items.map((item) => item.folderId);
    if (new Set(folderIds).size !== folderIds.length) throw new BadRequestException('เลือก folder ซ้ำกัน');

    const { connection, accessToken } = await this.ownConnection(user);

    const caseIds = [...new Set(items.map((item) => item.caseId))];
    const allowed = await this.prisma.case.findMany({
      where: { AND: [this.caseAccess.getCaseFilterForUser(user), { id: { in: caseIds } }] },
      select: { id: true },
    });
    if (allowed.length !== caseIds.length) throw new NotFoundException('ไม่พบคดีที่เลือก หรือคุณไม่มีสิทธิ์ในคดีนั้น');

    const folders = new Map((await this.folders.listMailFolders(accessToken)).map((folder) => [folder.id, folder]));
    for (const item of items) {
      const folder = folders.get(item.folderId);
      if (!folder) throw new BadRequestException('ไม่พบ folder นี้ในกล่องอีเมลของคุณ');
      if (folder.isSystem) {
        throw new BadRequestException(`ผูก "${folder.displayPath}" กับคดีไม่ได้ — เลือก folder ย่อยที่เก็บอีเมลของคดีนี้`);
      }
    }

    const existing = await this.prisma.caseExternalLink.findMany({
      where: { connectionId: connection.id, kind: OUTLOOK_FOLDER, externalId: { in: folderIds } },
      select: { externalId: true, caseId: true, displayPath: true },
    });
    const existingByFolder = new Map(existing.map((link) => [link.externalId, link]));
    for (const item of items) {
      const current = existingByFolder.get(item.folderId);
      if (current && current.caseId !== item.caseId) {
        throw new ConflictException(`folder "${current.displayPath}" ผูกกับคดีอื่นอยู่แล้ว — ยกเลิกการผูกเดิมก่อน`);
      }
    }

    const fresh = items.filter((item) => !existingByFolder.has(item.folderId));
    if (fresh.length === 0) return { linked: [] };

    const created = await this.prisma.$transaction(
      fresh.map((item) =>
        this.prisma.caseExternalLink.create({
          data: {
            firmId: user.firmId,
            caseId: item.caseId,
            connectionId: connection.id,
            kind: OUTLOOK_FOLDER,
            externalId: item.folderId,
            displayPath: folders.get(item.folderId)!.displayPath,
            linkedById: user.id,
          },
        }),
      ),
    );
    await this.prisma.auditLog.createMany({
      data: created.map((link) => ({
        firmId: user.firmId,
        userId: user.id,
        action: 'CASE_FOLDER_LINKED',
        metadata: { caseId: link.caseId, linkId: link.id, displayPath: link.displayPath },
      })),
    });

    // First sync = full backfill; can take minutes on a large folder, so it never blocks the confirm request.
    for (const link of created) {
      this.sync.syncLink(link.id).catch((error) => this.logger.warn(`Initial sync of folder link ${link.id} failed: ${error}`));
    }

    return {
      linked: created.map((link) => ({
        id: link.id,
        caseId: link.caseId,
        externalId: link.externalId,
        displayPath: link.displayPath,
      })),
    };
  }

  listForCase(caseId: string) {
    return this.prisma.caseExternalLink.findMany({
      where: { caseId, kind: OUTLOOK_FOLDER },
      select: {
        id: true,
        displayPath: true,
        lastSyncedAt: true,
        lastError: true,
        linkedById: true,
        connection: { select: { mailboxAddress: true, status: true } },
      },
      orderBy: { createdAt: 'asc' },
    });
  }

  /** Stops syncing; email already filed in the case stays there. */
  async unlink(user: AuthUser, linkId: string): Promise<{ unlinked: true }> {
    const link = await this.prisma.caseExternalLink.findFirst({
      where: { id: linkId, firmId: user.firmId, kind: OUTLOOK_FOLDER },
      select: { id: true, caseId: true, displayPath: true, linkedById: true, connection: { select: { userId: true } } },
    });
    if (!link) throw new NotFoundException('ไม่พบการผูก folder นี้');
    const canManage =
      link.linkedById === user.id ||
      link.connection.userId === user.id ||
      hasFirmCapability(user.firmRole, 'viewAllCases');
    if (!canManage) {
      throw new ForbiddenException('ยกเลิกการผูกได้เฉพาะผู้ที่ผูก เจ้าของกล่องอีเมล หรือผู้ดูแลสำนักงาน');
    }
    await this.prisma.caseExternalLink.delete({ where: { id: link.id } });
    await this.prisma.auditLog.create({
      data: {
        firmId: user.firmId,
        userId: user.id,
        action: 'CASE_FOLDER_UNLINKED',
        metadata: { caseId: link.caseId, linkId: link.id, displayPath: link.displayPath },
      },
    });
    return { unlinked: true };
  }
}
```

```ts
import { ConflictException, ForbiddenException, Injectable, NotFoundException } from '@nestjs/common';
import { AuthUser } from '@lawfirm/shared';
import { PrismaService } from '../prisma/prisma.module';
import { CaseAccessService } from '../common/services/case-access.service';
import { FileStorageService } from '../common/services/file-storage.service';
import { DocumentsService } from '../documents/documents.service';
import { DocumentCategory, EmailThreadStatus, Prisma } from '../generated/prisma';
import { MATCHABLE_CASE_SELECT, suggestCaseMatches, toMatchableCase } from '../outlook-integration/case-matching';

export type ThreadMatchReason = 'CONTACT_EMAIL' | 'OWN_REF' | 'CASE_NUMBER';

export interface ThreadCaseSuggestion {
  caseId: string;
  ownRef: string;
  title: string;
  reasons: ThreadMatchReason[];
}

const THREAD_REASON_RANK: Record<ThreadMatchReason, number> = { OWN_REF: 0, CASE_NUMBER: 1, CONTACT_EMAIL: 2 };
const MAX_SUGGESTIONS = 10;
const MAX_ADDRESSES = 20;
const MAX_CASE_THREADS = 200;

/**
 * Email ↔ case: manual thread linking (with exact-rule suggestions — no AI,
 * a person always confirms) and the case "อีเมล" tab. Every read goes through
 * getEmailThreadFilterForUser, so a thread is only visible here when the
 * Phase 0 visibility rule already allows it.
 */
@Injectable()
export class CaseEmailService {
  constructor(
    private prisma: PrismaService,
    private caseAccess: CaseAccessService,
    private documents: DocumentsService,
    private fileStorage: FileStorageService,
  ) {}

  private async threadWhere(user: AuthUser, extra: Prisma.EmailThreadWhereInput): Promise<Prisma.EmailThreadWhereInput> {
    return { AND: [await this.caseAccess.getEmailThreadFilterForUser(user), extra] };
  }

  private audit(user: AuthUser, action: 'EMAIL_THREAD_LINKED' | 'EMAIL_THREAD_UNLINKED', metadata: Record<string, string>) {
    return this.prisma.auditLog.create({ data: { firmId: user.firmId, userId: user.id, action, metadata } });
  }

  async suggestCasesForThread(user: AuthUser, threadId: string): Promise<ThreadCaseSuggestion[]> {
    const thread = await this.prisma.emailThread.findFirst({
      where: await this.threadWhere(user, { id: threadId }),
      select: {
        subject: true,
        fromAddress: true,
        mailboxConnection: { select: { mailboxAddress: true } },
        messages: { select: { fromAddress: true, toAddresses: true, ccAddresses: true } },
      },
    });
    if (!thread) throw new NotFoundException('ไม่พบอีเมลนี้');

    const ownMailbox = thread.mailboxConnection?.mailboxAddress.toLowerCase();
    const addresses = [
      ...new Set(
        [thread.fromAddress, ...thread.messages.flatMap((m) => [m.fromAddress, ...m.toAddresses, ...m.ccAddresses])]
          .map((address) => address?.trim().toLowerCase() ?? '')
          .filter((address) => address && address !== ownMailbox),
      ),
    ].slice(0, MAX_ADDRESSES);

    const caseFilter = this.caseAccess.getCaseFilterForUser(user);
    const emailIs = addresses.map((address) => ({ email: { equals: address, mode: 'insensitive' as const } }));
    const [contactCases, allCases] = await Promise.all([
      addresses.length
        ? this.prisma.case.findMany({
            where: {
              AND: [
                caseFilter,
                {
                  OR: [
                    { client: { OR: addresses.map((a) => ({ billingEmail: { equals: a, mode: 'insensitive' as const } })) } },
                    { client: { contacts: { some: { OR: emailIs } } } },
                    { participants: { some: { OR: emailIs } } },
                  ],
                },
              ],
            },
            select: { id: true, ownRef: true, title: true },
            take: MAX_SUGGESTIONS,
          })
        : Promise.resolve([] as { id: string; ownRef: string; title: string }[]),
      this.prisma.case.findMany({ where: caseFilter, select: MATCHABLE_CASE_SELECT }),
    ]);

    const byCase = new Map<string, ThreadCaseSuggestion>();
    const add = (c: { id: string; ownRef: string; title: string }, reason: ThreadMatchReason) => {
      const entry = byCase.get(c.id) ?? { caseId: c.id, ownRef: c.ownRef, title: c.title, reasons: [] };
      if (!entry.reasons.includes(reason)) entry.reasons.push(reason);
      byCase.set(c.id, entry);
    };
    for (const match of suggestCaseMatches(thread.subject, allCases.map(toMatchableCase))) {
      if (match.reason === 'CLIENT_NAME') continue; // names in subjects are too noisy — sender/recipient emails cover the client
      add({ id: match.caseId, ownRef: match.ownRef, title: match.title }, match.reason);
    }
    for (const c of contactCases) add(c, 'CONTACT_EMAIL');

    const best = (s: ThreadCaseSuggestion) => Math.min(...s.reasons.map((r) => THREAD_REASON_RANK[r]));
    return [...byCase.values()]
      .sort((a, b) => b.reasons.length - a.reasons.length || best(a) - best(b) || a.ownRef.localeCompare(b.ownRef))
      .slice(0, MAX_SUGGESTIONS);
  }

  async linkThread(user: AuthUser, threadId: string, caseId: string): Promise<{ threadId: string; caseId: string }> {
    const thread = await this.prisma.emailThread.findFirst({
      where: await this.threadWhere(user, { id: threadId }),
      select: { id: true, caseId: true, intakeId: true },
    });
    if (!thread) throw new NotFoundException('ไม่พบอีเมลนี้');
    if (thread.caseId === caseId) return { threadId: thread.id, caseId };
    if (thread.caseId) throw new ConflictException('อีเมลนี้ผูกกับคดีอื่นอยู่แล้ว — ยกเลิกการผูกเดิมก่อน');
    if (!(await this.caseAccess.canAccessCase(user, caseId))) {
      throw new NotFoundException('ไม่พบคดี หรือคุณไม่มีสิทธิ์ในคดีนี้');
    }

    await this.prisma.emailThread.update({
      where: { id: thread.id },
      data: { caseId, status: EmailThreadStatus.LINKED },
    });
    await this.audit(user, 'EMAIL_THREAD_LINKED', { threadId: thread.id, caseId });
    return { threadId: thread.id, caseId };
  }

  async unlinkThread(user: AuthUser, threadId: string): Promise<{ unlinked: boolean }> {
    const thread = await this.prisma.emailThread.findFirst({
      where: await this.threadWhere(user, { id: threadId }),
      select: { id: true, caseId: true, intakeId: true },
    });
    if (!thread) throw new NotFoundException('ไม่พบอีเมลนี้');
    if (!thread.caseId) return { unlinked: false };
    if (!(await this.caseAccess.canAccessCase(user, thread.caseId))) {
      throw new ForbiddenException('ยกเลิกการผูกได้เฉพาะผู้ที่มีสิทธิ์ในคดีนี้');
    }

    await this.prisma.emailThread.update({
      where: { id: thread.id },
      data: { caseId: null, status: thread.intakeId ? EmailThreadStatus.LINKED : EmailThreadStatus.PENDING_INTAKE },
    });
    await this.audit(user, 'EMAIL_THREAD_UNLINKED', { threadId: thread.id, caseId: thread.caseId });
    return { unlinked: true };
  }

  async listCaseThreads(user: AuthUser, caseId: string) {
    const threads = await this.prisma.emailThread.findMany({
      where: await this.threadWhere(user, { caseId }),
      select: {
        id: true,
        subject: true,
        fromName: true,
        fromAddress: true,
        lastMessageAt: true,
        messages: { select: { _count: { select: { attachments: true } } } },
      },
      orderBy: { lastMessageAt: 'desc' },
      take: MAX_CASE_THREADS,
    });
    return threads.map((thread) => ({
      id: thread.id,
      subject: thread.subject,
      fromName: thread.fromName,
      fromAddress: thread.fromAddress,
      lastMessageAt: thread.lastMessageAt,
      messageCount: thread.messages.length,
      attachmentCount: thread.messages.reduce((sum, m) => sum + m._count.attachments, 0),
    }));
  }

  async getCaseThread(user: AuthUser, caseId: string, threadId: string) {
    const thread = await this.prisma.emailThread.findFirst({
      where: await this.threadWhere(user, { id: threadId, caseId }),
      select: {
        id: true,
        subject: true,
        caseId: true,
        lastMessageAt: true,
        messages: {
          orderBy: { receivedAt: 'asc' },
          select: {
            id: true,
            direction: true,
            fromName: true,
            fromAddress: true,
            toAddresses: true,
            ccAddresses: true,
            bodyText: true,
            bodyHtml: true,
            receivedAt: true,
            attachments: {
              orderBy: { createdAt: 'asc' },
              select: { id: true, filename: true, mimeType: true, createdAt: true },
            },
          },
        },
      },
    });
    if (!thread) throw new NotFoundException('ไม่พบอีเมลนี้ในคดี');
    return thread;
  }

  /** "บันทึกไฟล์แนบเข้าเอกสารคดี" — idempotent per attachment via a tag, so a double click saves once. */
  async saveAttachmentToCase(user: AuthUser, caseId: string, attachmentId: string) {
    const attachment = await this.prisma.emailAttachment.findFirst({
      where: { id: attachmentId, message: { thread: await this.threadWhere(user, { caseId }) } },
      select: { id: true, filename: true, mimeType: true, storagePath: true },
    });
    if (!attachment) throw new NotFoundException('ไม่พบไฟล์แนบนี้ในคดี');

    const tag = `email-attachment:${attachment.id}`;
    const existing = await this.prisma.document.findFirst({
      where: { caseId, tags: { has: tag } },
      select: { id: true, filename: true },
    });
    if (existing) return { document: existing, alreadySaved: true };

    const buffer = await this.fileStorage.getBuffer(attachment.storagePath);
    const document = await this.documents.createFromBuffer(user, caseId, {
      filename: attachment.filename,
      buffer,
      mimeType: attachment.mimeType,
      category: DocumentCategory.CORRESPONDENCE,
      tags: ['อีเมล', tag],
    });
    return { document: { id: document.id, filename: document.filename }, alreadySaved: false };
  }
}
```

```ts
import { Body, Controller, Delete, Get, Param, Post, UseGuards } from '@nestjs/common';
import { AuthUser } from '@lawfirm/shared';
import { JwtAuthGuard } from '../common/guards/jwt-auth.guard';
import { CaseAccessGuard } from '../common/guards/case-access.guard';
import { CurrentUser } from '../common/decorators/current-user.decorator';
import { CaseEmailService } from './case-email.service';
import { LinkThreadToCaseDto } from './dto/email-intake.dto';

@Controller()
@UseGuards(JwtAuthGuard)
export class CaseEmailController {
  constructor(private service: CaseEmailService) {}

  @Get('email-intake/threads/:threadId/case-suggestions')
  suggestions(@CurrentUser() user: AuthUser, @Param('threadId') threadId: string) {
    return this.service.suggestCasesForThread(user, threadId);
  }

  @Post('email-intake/threads/:threadId/case-link')
  link(@CurrentUser() user: AuthUser, @Param('threadId') threadId: string, @Body() dto: LinkThreadToCaseDto) {
    return this.service.linkThread(user, threadId, dto.caseId);
  }

  @Delete('email-intake/threads/:threadId/case-link')
  unlink(@CurrentUser() user: AuthUser, @Param('threadId') threadId: string) {
    return this.service.unlinkThread(user, threadId);
  }

  @Get('cases/:caseId/email/threads')
  @UseGuards(CaseAccessGuard)
  listCaseThreads(@CurrentUser() user: AuthUser, @Param('caseId') caseId: string) {
    return this.service.listCaseThreads(user, caseId);
  }

  @Get('cases/:caseId/email/threads/:threadId')
  @UseGuards(CaseAccessGuard)
  getCaseThread(@CurrentUser() user: AuthUser, @Param('caseId') caseId: string, @Param('threadId') threadId: string) {
    return this.service.getCaseThread(user, caseId, threadId);
  }

  @Post('cases/:caseId/email/attachments/:attachmentId/save-to-documents')
  @UseGuards(CaseAccessGuard)
  saveAttachment(@CurrentUser() user: AuthUser, @Param('caseId') caseId: string, @Param('attachmentId') attachmentId: string) {
```

```ts
/**
 * Email HTML is untrusted. The real boundary is the sandboxed iframe in
 * EmailHtmlFrame (no allow-scripts, no allow-same-origin) plus the CSP below;
 * this strip is defence in depth.
 * ponytail: regex strip, not a parser — entity-obfuscated payloads get past it but still hit the sandbox/CSP; swap for DOMPurify if HTML ever renders outside the sandbox.
 */
const CSP = "default-src 'none'; img-src data:; style-src 'unsafe-inline'; font-src data:";

export function sanitizeEmailHtml(html: string): string {
  return html
    .replace(/<script\b[\s\S]*?<\/script\s*>/gi, '')
    .replace(/<(iframe|object|embed|frameset|frame|noscript)\b[\s\S]*?<\/\1\s*>/gi, '')
    .replace(/<\/?(script|iframe|object|embed|frame|frameset|link|meta|base|form|input|button|textarea|select)\b[^>]*>/gi, '')
    .replace(/\son[a-z]+\s*=\s*("[^"]*"|'[^']*'|[^\s>]+)/gi, '')
    .replace(/\b(href|src|action|xlink:href)\s*=\s*(["']?)\s*javascript:/gi, '$1=$2#blocked:');
}

export function buildEmailSrcDoc(html: string): string {
  return (
    '<!doctype html><html><head><meta charset="utf-8">' +
    `<meta http-equiv="Content-Security-Policy" content="${CSP}">` +
    '<base target="_blank">' +
    '<style>body{font-family:system-ui,sans-serif;font-size:14px;line-height:1.5;margin:8px;overflow-wrap:anywhere;color:#111;background:#fff}img{max-width:100%;height:auto}</style>' +
    `</head><body>${sanitizeEmailHtml(html)}</body></html>`
  );
}
```

```tsx
                                <Download className="h-4 w-4" /> ดาวน์โหลด

                              </Button>

                        </ul>

                    </article>
                  ))}
                  <p className="text-xs text-muted-foreground">รูปภาพจากภายนอกในอีเมลถูกบล็อกเพื่อความปลอดภัย</p>
                </CardContent>
              </Card>
            )}
          </div>
        </div>
      )}
    </div>
  );
}
```

```tsx
'use client';

import { useEffect, useMemo, useState } from 'react';
import Link from 'next/link';
import { Link2 } from 'lucide-react';
import { useAuth } from '@/lib/auth';
import { api, ApiError, type CaseListItem, type ThreadCaseSuggestion } from '@/lib/api';
import { OTHER_CASE, THREAD_REASON_LABELS } from '@/lib/case-email';
import { caseTabHref } from '@/lib/case-tabs';
import { Button } from '@/components/ui/button';
import { Card, CardContent } from '@/components/ui/card';

/** "ผูกกับคดี": suggestions are shown, never pre-selected — the person picks and confirms. */
export function LinkThreadToCaseCard({
  threadId,
  linkedCase,
  onChanged,
}: {
  threadId: string;
  linkedCase: { id: string; ownRef: string; title: string } | null;
  onChanged: () => void;
}) {
  const { token } = useAuth();
  const [open, setOpen] = useState(false);
  const [suggestions, setSuggestions] = useState<ThreadCaseSuggestion[] | null>(null);
  const [cases, setCases] = useState<CaseListItem[]>([]);
  const [chosenId, setChosenId] = useState('');
  const [otherRef, setOtherRef] = useState('');
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState('');

  useEffect(() => {
    if (!open || !token) return;
    let active = true;
    Promise.all([api.getThreadCaseSuggestions(token, threadId), api.getCases(token).catch(() => [] as CaseListItem[])])
      .then(([nextSuggestions, nextCases]) => {
        if (!active) return;
        setSuggestions(nextSuggestions);
        setCases(nextCases);
      })
      .catch((e) => active && setError(e instanceof ApiError ? e.message : 'โหลดคดีที่แนะนำไม่สำเร็จ'));
    return () => {
      active = false;
    };
  }, [open, token, threadId]);

  const caseIdByRef = useMemo(() => new Map(cases.map((c) => [c.ownRef, c.id])), [cases]);
  const targetCaseId = chosenId === OTHER_CASE ? caseIdByRef.get(otherRef.trim()) ?? '' : chosenId;

  const submit = async (action: (t: string) => Promise<unknown>, fallback: string) => {
    if (!token) return;
    setBusy(true);
    setError('');
    try {
      await action(token);
      setOpen(false);
      setChosenId('');
      setOtherRef('');
      onChanged();
    } catch (e) {
      setError(e instanceof ApiError ? e.message : fallback);
    } finally {
      setBusy(false);
    }
  };

  if (linkedCase) {
    return (
      <Card className="mb-4">
        <CardContent className="flex flex-wrap items-center justify-between gap-3 p-4">
          <p className="text-sm">
            ผูกกับคดี{' '}
            <Link href={caseTabHref(linkedCase.id, 'email')} className="font-medium text-primary hover:underline">
              {linkedCase.ownRef} · {linkedCase.title}
            </Link>
          </p>
          <Button
            size="sm"
            variant="outline"
            disabled={busy}
            onClick={() => {
              if (window.confirm('ยกเลิกการผูกอีเมลนี้กับคดี? อีเมลจะกลับเข้าคิว')) {
                submit((t) => api.unlinkThreadFromCase(t, threadId), 'ยกเลิกการผูกไม่สำเร็จ');
              }
            }}
          >
            ยกเลิกการผูก
          </Button>
          {error && <p className="w-full text-sm text-destructive">{error}</p>}
        </CardContent>
      </Card>
    );
  }

  return (
    <Card className="mb-4">
      <CardContent className="space-y-3 p-4">
        {!open ? (
          <Button size="sm" variant="outline" onClick={() => setOpen(true)}>
            <Link2 className="h-4 w-4" /> ผูกกับคดี
          </Button>
        ) : (
          <fieldset className="space-y-2">
            <legend className="text-sm font-medium">เลือกคดีที่จะผูกอีเมลนี้</legend>
            {suggestions === null && !error && <p className="text-sm text-muted-foreground">กำลังหาคดีที่เกี่ยวข้อง…</p>}
            {suggestions?.length === 0 && (
              <p className="text-sm text-muted-foreground">ไม่พบคดีที่ตรงกับผู้ส่ง/ผู้รับหรือหัวเรื่อง — เลือกคดีเองด้านล่าง</p>
            )}
            {suggestions?.map((s) => (
              <label key={s.caseId} className="flex items-start gap-2 text-sm">
                <input
                  type="radio"
                  name="link-case"
                  value={s.caseId}
                  checked={chosenId === s.caseId}
                  onChange={() => setChosenId(s.caseId)}
                  className="mt-1"
                />
                <span>
                  <span className="font-medium">{s.ownRef}</span> · {s.title}
                  <span className="block text-xs text-muted-foreground">
                    {s.reasons.map((r) => THREAD_REASON_LABELS[r]).join(' · ')}
                  </span>
                </span>
              </label>
            ))}
            <label className="flex items-center gap-2 text-sm">
              <input
                type="radio"
                name="link-case"
                value={OTHER_CASE}
                checked={chosenId === OTHER_CASE}
                onChange={() => setChosenId(OTHER_CASE)}
              />
              คดีอื่น
            </label>
            {chosenId === OTHER_CASE && (
              <>
                <input
                  list="link-case-refs"
                  value={otherRef}
                  onChange={(e) => setOtherRef(e.target.value)}
                  placeholder="พิมพ์เลขอ้างอิงคดี"
                  aria-label="เลขอ้างอิงคดี"
                  className="w-full rounded-md border bg-background px-2 py-1.5 text-sm"
                />
                <datalist id="link-case-refs">
                  {cases.map((c) => (
                    <option key={c.id} value={c.ownRef}>
                      {c.title}
                    </option>
                  ))}
                </datalist>
              </>
            )}
            <div className="flex gap-2">
              <Button
                size="sm"
                disabled={!targetCaseId || busy}
                onClick={() => submit((t) => api.linkThreadToCase(t, threadId, targetCaseId), 'ผูกกับคดีไม่สำเร็จ')}
              >
                {busy ? 'กำลังผูก…' : 'ยืนยันผูกกับคดีนี้'}
              </Button>
              <Button size="sm" variant="outline" onClick={() => setOpen(false)}>
                ยกเลิก
              </Button>
            </div>
          </fieldset>
        )}
        {error && <p className="text-sm text-destructive">{error}</p>}
      </CardContent>
    </Card>
  );
}
```

```tsx
import { ArrowLeft } from 'lucide-react';
import { useAuth } from '@/lib/auth';
import { api, ApiError, type CaseListItem, type OutlookFolderRow } from '@/lib/api';
import { defaultFolderCaseId, MATCH_REASON_LABELS, OTHER_CASE } from '@/lib/case-email';
import { Button } from '@/components/ui/button';
import { Card, CardContent } from '@/components/ui/card';
import { PageLoading } from '@/components/ui/misc';

type Notice = { type: 'success' | 'error'; text: string } | null;

/** ตั้งค่า → Microsoft 365 → จับคู่ folder: suggestions are a pre-fill only; nothing links until "ยืนยัน". */
export default function OutlookFolderMatchPage() {
  const { token } = useAuth();
  const [rows, setRows] = useState<OutlookFolderRow[] | null>(null);
  const [cases, setCases] = useState<CaseListItem[]>([]);
  const [selection, setSelection] = useState<Record<string, string>>({});
  const [otherRefs, setOtherRefs] = useState<Record<string, string>>({});
  const [onlySuggested, setOnlySuggested] = useState(true);
  const [loadError, setLoadError] = useState('');
  const [notice, setNotice] = useState<Notice>(null);
  const [saving, setSaving] = useState(false);

  const load = useCallback(async () => {
    if (!token) return;
    setLoadError('');
    try {
      const [folderRows, caseList] = await Promise.all([
        api.getOutlookFolders(token),
        api.getCases(token).catch(() => [] as CaseListItem[]),
      ]);
      setRows(folderRows);
      setCases(caseList);
      setSelection(Object.fromEntries(folderRows.map((row) => [row.folderId, defaultFolderCaseId(row)])));
      setOtherRefs({});
    } catch (e) {
      setLoadError(e instanceof ApiError ? e.message : 'ดึงรายชื่อ folder จาก Outlook ไม่สำเร็จ');
    }
  }, [token]);

  useEffect(() => {
    load();
  }, [load]);

  const caseIdByRef = useMemo(() => new Map(cases.map((c) => [c.ownRef, c.id])), [cases]);
  const resolvedCaseId = (folderId: string) =>
    selection[folderId] === OTHER_CASE ? caseIdByRef.get((otherRefs[folderId] ?? '').trim()) ?? '' : selection[folderId] ?? '';

  const pending = (rows ?? [])
    .filter((row) => !row.link)
    .map((row) => ({ folderId: row.folderId, caseId: resolvedCaseId(row.folderId) }))
    .filter((item) => item.caseId);

  const visible = (rows ?? []).filter(
    (row) => !onlySuggested || row.link || row.suggestions.length > 0 || selection[row.folderId],
  );

  const confidence = (row: OutlookFolderRow) => {
    if (row.link) return 'ผูกแล้ว';
    if (selection[row.folderId] === OTHER_CASE) return 'เลือกเอง';
    const hit = row.suggestions.find((s) => s.caseId === selection[row.folderId]);
    return hit ? MATCH_REASON_LABELS[hit.reason] : '—';
  };

  const confirm = async () => {
    if (!token || pending.length === 0) return;
    setSaving(true);
    setNotice(null);
    try {
      const result = await api.linkOutlookFolders(token, pending);
      setNotice({ type: 'success', text: `ผูกแล้ว ${result.linked.length} folder — ระบบกำลังดึงอีเมลย้อนหลังเข้าแต่ละคดี` });
      await load();
    } catch (e) {
      setNotice({ type: 'error', text: e instanceof ApiError ? e.message : 'ผูก folder ไม่สำเร็จ' });
    } finally {
      setSaving(false);
    }
  };

  const unlink = async (linkId: string) => {
    if (!token || !window.confirm('ยกเลิกการผูก folder นี้? อีเมลที่ดึงเข้าคดีแล้วจะยังอยู่ในคดี')) return;
    try {
      await api.unlinkOutlookFolder(token, linkId);
      await load();
    } catch (e) {
      setNotice({ type: 'error', text: e instanceof ApiError ? e.message : 'ยกเลิกการผูกไม่สำเร็จ' });
    }
  };

  return (
    <div className="space-y-4">
      <Link href="/settings" className="inline-flex items-center gap-1 text-sm text-primary hover:underline">
        <ArrowLeft className="h-4 w-4" /> กลับไปหน้าตั้งค่า
      </Link>
      <div>
        <h1 className="text-2xl font-bold tracking-tight">จับคู่ folder Outlook กับคดี</h1>
        <p className="mt-1 text-sm text-muted-foreground">
          ระบบแนะนำคดีจากเลขอ้างอิง เลขคดีดำ/แดง และชื่อลูกความในชื่อ folder — ไม่มีการผูกจนกว่าคุณจะกด &quot;ยืนยัน&quot;.
          อีเมลใน folder ที่ผูกจะเข้าแท็บอีเมลของคดี และทุกคนที่มีสิทธิ์ในคดีจะเห็น
        </p>
      </div>

      {loadError && (
        <p className="rounded-lg bg-destructive/10 p-3 text-sm text-destructive">
          {loadError}{' '}
          <Link href="/settings" className="underline">
            ไปหน้าตั้งค่า
          </Link>
        </p>
      )}
      {notice && (
        <p
          role="status"
          className={`rounded-lg p-3 text-sm ${notice.type === 'success' ? 'bg-green-50 text-green-700' : 'bg-destructive/10 text-destructive'}`}
        >
          {notice.text}
        </p>
      )}

      {!rows && !loadError && <PageLoading title="กำลังดึงรายชื่อ folder จาก Outlook" lines={4} />}

      {rows && (
        <Card>
          <CardContent className="space-y-3 p-4">
            <label className="flex items-center gap-2 text-sm">
              <input type="checkbox" checked={onlySuggested} onChange={(e) => setOnlySuggested(e.target.checked)} />
              แสดงเฉพาะ folder ที่มีคดีแนะนำหรือผูกแล้ว ({visible.length}/{rows.length})
            </label>
            <div className="overflow-x-auto">
              <table className="w-full min-w-[640px] text-sm">
                <thead>
                  <tr className="text-left text-xs text-muted-foreground">
                    <th className="py-2 pr-3 font-medium">folder</th>
                    <th className="py-2 pr-3 font-medium">คดี</th>
                    <th className="py-2 font-medium">ความมั่นใจ</th>
                  </tr>
                </thead>
                <tbody>
                  {visible.map((row) => (
                    <tr key={row.folderId} className="border-t align-top">
                      <td className="break-all py-2 pr-3">{row.displayPath}</td>
                      <td className="py-2 pr-3">
                        {row.link ? (
                          <div className="flex flex-wrap items-center gap-2">
                            <span>
                              ผูกกับ <strong>{row.link.ownRef}</strong> · {row.link.title}
                            </span>
                            <Button size="sm" variant="outline" onClick={() => unlink(row.link!.id)}>
                              ยกเลิก
                            </Button>
                          </div>
                        ) : (
                          <div className="space-y-1">
                            <select
                              aria-label={`คดีสำหรับ ${row.displayPath}`}
                              value={selection[row.folderId] ?? ''}
                              onChange={(e) => setSelection((s) => ({ ...s, [row.folderId]: e.target.value }))}
                              className="w-full rounded-md border bg-background px-2 py-1.5 text-sm"
                            >
                              <option value="">— ข้าม —</option>
                              {row.suggestions.map((s) => (
                                <option key={s.caseId} value={s.caseId}>
                                  {s.ownRef} · {s.title}
                                </option>
                              ))}
                              <option value={OTHER_CASE}>เลือกคดีอื่น…</option>
                            </select>
                            {selection[row.folderId] === OTHER_CASE && (
                              <input
                                list="folder-case-refs"
                                value={otherRefs[row.folderId] ?? ''}
                                onChange={(e) => setOtherRefs((r) => ({ ...r, [row.folderId]: e.target.value }))}
                                placeholder="พิมพ์เลขอ้างอิงคดี"
                                aria-label={`เลขอ้างอิงคดีสำหรับ ${row.displayPath}`}
                                className="w-full rounded-md border bg-background px-2 py-1.5 text-sm"
                              />
                            )}
                          </div>
                        )}
                      </td>
                      <td className="py-2 text-xs text-muted-foreground">{confidence(row)}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
              <datalist id="folder-case-refs">
                {cases.map((c) => (
                  <option key={c.id} value={c.ownRef}>
                    {c.title}
                  </option>
                ))}
              </datalist>
            </div>
            <Button disabled={pending.length === 0 || saving} onClick={confirm}>
              {saving ? 'กำลังผูก…' : `ยืนยันการผูก (${pending.length} folder)`}
            </Button>
          </CardContent>
        </Card>
      )}
    </div>
  );
}
```
